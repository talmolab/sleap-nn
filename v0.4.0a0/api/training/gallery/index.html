    gallery

    sleap_nn.training.gallery ¶

    Prediction galleries for W&B: panel selection, prediction and drawing.

    Used by :class:~sleap_nn.training.callbacks.WandBGalleryCallback. Everything here is pure (numpy / OpenCV / torch, no Lightning state), so it can be tested on its own:

    • :func:select_panel picks a fixed, stratified validation panel from per-sample metadata (no image decoding).
    • :func:predict_boxes / :func:predict_keypoints / :func:predict_instances run one no-grad forward over a batch of panel images with the model's own decoders.
    • :func:bbox_wandb_image builds a wandb.Image with native box overlays (ground truth and predictions, scores attached so the W&B viewer can filter them); :func:draw_keypoints draws keypoints and skeleton edges with OpenCV; :func:draw_instances draws the grouped instances of a whole frame (bottom-up models) with sleap-io's renderer (Skia).

    Only the *_wandb_image helpers need W&B; the rest runs without it.

    def bbox_wandb_image(
    image: Optional[np.ndarray],
    gt_boxes: np.ndarray,
    caption: str,
    categories: Sequence[str],
    pred_boxes: Optional[np.ndarray] = None,
    pred_scores: Optional[np.ndarray] = None,
    gt_classes: Optional[np.ndarray] = None,
    long_side: int = 640,
    max_preds: int = 50,
    cached: Optional[Tuple[str, float]] = None,
):
    """A JPEG ``wandb.Image`` of a frame with native W&B box overlays.

    Args:
        image: ``(H, W, 3)`` uint8 RGB frame (model-input pixels); unused with
            ``cached``.
        gt_boxes: ``(n, 4)`` xyxy ground-truth boxes in ``image`` pixels.
        caption: Image caption.
        categories: Class names (``class_id`` indexes them).
        pred_boxes: Predicted ``(m, 4)`` boxes, or ``None`` (training inputs show
            their targets only).
        pred_scores: Scores of ``pred_boxes``; the ``max_preds`` highest are kept.
        gt_classes: Ground-truth class ids (default all 0).
        long_side: The image is downscaled to this long side before encoding.
        max_preds: Maximum predicted boxes per image (highest scores).
        cached: ``(jpeg_path, scale)`` of the frame already downscaled and encoded
            by :func:`encode_display_jpeg` (validation panels are encoded once).

    Returns:
        A ``wandb.Image``.
    """
    import wandb

    if cached is not None:
        data, s = cached
    else:
        data, s = downscale(image, long_side)
    class_labels = {i: str(c) for i, c in enumerate(categories or ["object"])}
    boxes = {
        "ground_truth": {
            "box_data": _box_entries(
                gt_boxes, s, gt_classes, names=list(class_labels.values())
            ),
            "class_labels": class_labels,
        }
    }
    if pred_boxes is not None:
        pred_boxes = np.asarray(pred_boxes).reshape(-1, 4)
        pred_scores = np.asarray(pred_scores).reshape(-1)
        keep = np.argsort(-pred_scores, kind="stable")[:max_preds]
        boxes["predictions"] = {
            "box_data": _box_entries(
                pred_boxes[keep], s, None, scores=pred_scores[keep]
            ),
            "class_labels": class_labels,
        }
    if cached is not None:
        return wandb.Image(data, caption=caption, boxes=boxes)
    return wandb.Image(data, caption=caption, boxes=boxes, file_type="jpg")

    def box_hardness(
    boxes_per_sample: Sequence[np.ndarray],
    groups: Optional[np.ndarray] = None,
    min_rel_side: float = MIN_HARD_REL_SIDE,
) -> np.ndarray:
    """Detector hardness: most boxes first, then the smallest box (``sqrt`` area).

    Only boxes at least ``min_rel_side`` times the median box side of their group
    (file) count (:data:`MIN_HARD_REL_SIDE`); a frame with none is not eligible.

    Args:
        boxes_per_sample: Per sample, its ``(n, 4)`` xyxy ground-truth boxes.
        groups: ``(n_samples,)`` group (file) of every sample; ``None``: one group.
        min_rel_side: Eligibility floor, as a fraction of the group's median side.

    Returns:
        ``(n_samples,)`` scores, higher = harder; ``-inf`` = not eligible.
    """
    n = len(boxes_per_sample)
    groups = np.zeros(n, dtype=int) if groups is None else np.asarray(groups)
    sides = []
    for b in boxes_per_sample:
        b = np.asarray(b, dtype=np.float64).reshape(-1, 4)
        sides.append(_sides(b[np.isfinite(b).all(axis=1)]))
    flat = np.concatenate(sides) if n else np.zeros(0)
    flat_groups = np.repeat(groups, [len(s) for s in sides]) if n else flat
    medians = dict(
        zip(flat_groups.tolist(), _group_medians(flat, flat_groups).tolist())
    )
    out = np.full(n, -np.inf)
    for i, (s, g) in enumerate(zip(sides, groups.tolist())):
        med = medians.get(g, np.nan)
        s = s[s >= min_rel_side * med] if np.isfinite(med) else s[:0]
        if len(s):
            out[i] = len(s) * 1e6 - s.min()
    return out

    def caption(
    name: str,
    lab: Optional[str],
    video_idx: int,
    frame_idx: int,
    extra: str = "",
    role: Optional[str] = None,
) -> str:
    """``"<dataset> | <lab> | v<video> f<frame> | <role> | <extra>"``."""
    parts = [str(name)]
    if lab:
        parts.append(str(lab))
    parts.append(f"v{int(video_idx)} f{int(frame_idx)}")
    if role:
        parts.append(role)
    if extra:
        parts.append(extra)
    return " | ".join(parts)

    def downscale(image: np.ndarray, long_side: int) -> Tuple[np.ndarray, float]:
    """Shrink ``image`` so its long side is at most ``long_side``; ``(image, scale)``."""
    h, w = image.shape[:2]
    s = min(1.0, float(long_side) / max(h, w))
    if s >= 1.0:
        return image, 1.0
    size = (max(1, round(w * s)), max(1, round(h * s)))
    return cv2.resize(image, size, interpolation=cv2.INTER_AREA), s

    def draw_instances(
    image: np.ndarray,
    edges: Sequence[Tuple[int, int]] = (),
    gt: Optional[np.ndarray] = None,
    pred: Optional[np.ndarray] = None,
    long_side: int = 640,
) -> np.ndarray:
    """Draw the animals of a whole frame with sleap-io's renderer (Skia).

    With predictions: every predicted instance in its own color (nodes and
    skeleton), on top of the ground truth drawn as thin green edges and hollow
    green circles. Without predictions (training targets): the ground-truth
    instances, each in its own color.

    Args:
        image: ``(H, W, 3)`` uint8 RGB frame (not modified).
        edges: Skeleton edges as ``(src, dst)`` node index pairs.
        gt: ``(n, K, 2)`` ground-truth instances in frame pixels (NaN = not
            labeled), or ``None``.
        pred: ``(m, K, 2)`` predicted instances, or ``None``.
        long_side: The frame is downscaled so its long side is at most this.

    Returns:
        A new ``(h, w, 3)`` uint8 image at the display size.
    """
    import skia
    from sleap_io.rendering.core import render_frame

    h, w = image.shape[:2]
    scale = min(1.0, float(long_side) / max(h, w))
    edges = [(int(a), int(b)) for a, b in edges]

    def finite(instances):
        if instances is None:
            return []
        arr = np.asarray(instances, dtype=np.float64)
        arr = arr.reshape(-1, arr.shape[-2], 2) if arr.size else arr.reshape(0, 0, 2)
        return [p for p in arr if np.isfinite(p).all(axis=-1).any()]

    gt_list, pred_list = finite(gt), finite(pred)
    if pred is None:
        drawn, under = gt_list, []
    else:
        drawn, under = pred_list, gt_list

    def draw_ground_truth(ctx):
        paint = skia.Paint(
            Color=skia.Color(*_GT_COLOR),
            AntiAlias=True,
            Style=skia.Paint.kStroke_Style,
            StrokeWidth=1.0,
        )
        for pts in under:
            xy = pts * ctx.scale
            ok = np.isfinite(xy).all(axis=-1)
            for a, b in edges:
                if a < len(xy) and b < len(xy) and ok[a] and ok[b]:
                    ctx.canvas.drawLine(*map(float, xy[a]), *map(float, xy[b]), paint)
            for (x, y), good in zip(xy, ok):
                if good:
                    ctx.canvas.drawCircle(float(x), float(y), 4.0, paint)

    out = render_frame(
        np.ascontiguousarray(image),
        drawn,
        edge_inds=edges,
        node_names=[],
        color_by="instance",
        palette="distinct",
        marker_size=2.5,
        line_width=1.5,
        scale=scale,
        pre_render_callback=draw_ground_truth if under else None,
    )
    return np.ascontiguousarray(out)

    """Draw ground truth and predicted keypoints of one crop with OpenCV.

    Only ``nodes`` (the crop's own skeleton's channels) are drawn. With predictions:
    the predicted skeleton (amber edges, a dot per node in its own color, a cross
    where the peak value is below ``min_value``), ground truth as hollow green
    circles, and a red line from each ground-truth point to its prediction. Without
    predictions (training targets): the ground-truth skeleton in green.

    Args:
        image: ``(H, W, 3)`` uint8 RGB crop (not modified).
        nodes: Model channel indices to draw.
        edges: ``(src, dst)`` model channel index pairs.
        gt: ``(K, 2)`` ground truth in crop px (NaN = none), or ``None``.
        pred: ``(K, 2)`` predictions in crop px, or ``None``.
        pred_values: ``(K,)`` peak values, or ``None``.
        min_value: Peak value below which a prediction is drawn as a cross.

    Returns:
        A new ``(H, W, 3)`` uint8 image.
    """
    out = np.ascontiguousarray(image.copy())
    nodes = [int(n) for n in nodes]
    node_set = set(nodes)
    color = {
        n: tuple(int(c) for c in _PALETTE[i % len(_PALETTE)])
        for i, n in enumerate(nodes)
    }

    def pt(a, n):
        if a is None:
            return None
        p = a[n]
        if not np.isfinite(p).all():
            return None
        return (int(round(float(p[0]))), int(round(float(p[1]))))

    aa = cv2.LINE_AA
    skel, skel_color = (pred, _PRED_EDGE_COLOR) if pred is not None else (gt, _GT_COLOR)
    for a, b in edges:
        if a in node_set and b in node_set:
            pa, pb = pt(skel, a), pt(skel, b)
            if pa and pb:
                cv2.line(out, pa, pb, skel_color, 1, aa)
    for n in nodes:
        g, p = pt(gt, n), pt(pred, n)
        if g and p:
            cv2.line(out, g, p, _ERR_COLOR, 1, aa)
    for n in nodes:
        g, p = pt(gt, n), pt(pred, n)
        if g:
            cv2.circle(out, g, 4 if pred is not None else 2, _GT_COLOR, 1, aa)
        if p:
            low = pred_values is not None and not (float(pred_values[n]) >= min_value)
            if low:
                cv2.drawMarker(out, p, color[n], cv2.MARKER_TILTED_CROSS, 6, 1, aa)
            else:
                cv2.circle(out, p, 2, color[n], -1, aa)
    return out

    def instance_error(
    gt: np.ndarray, pred: np.ndarray, nodes: Optional[Sequence[int]] = None
) -> float:
    """Median node error (px) after greedily pairing each ground-truth instance.

    Pairs are taken in order of increasing mean node distance (over the ``nodes``
    both have), each instance used once; unpaired instances do not count.

    Args:
        gt: ``(n, K, 2)`` ground-truth instances.
        pred: ``(m, K, 2)`` predicted instances.
        nodes: Node indices to compare; ``None``: all.

    Returns:
        The median distance over the nodes of all pairs, ``nan`` without a pair.
    """
    gt = np.asarray(gt, dtype=np.float64).reshape(-1, *np.shape(gt)[-2:])
    pred = np.asarray(pred, dtype=np.float64)
    if gt.shape[0] == 0 or pred.size == 0:
        return float("nan")
    pred = pred.reshape(-1, *pred.shape[-2:])
    if nodes is not None:
        nodes = np.asarray(list(nodes), dtype=int)
        gt, pred = gt[:, nodes], pred[:, nodes]
    d = np.linalg.norm(gt[:, None] - pred[None], axis=-1)  # (n, m, K)
    ok = np.isfinite(d)
    count = ok.sum(-1)
    mean = np.where(
        count > 0, np.where(ok, d, 0.0).sum(-1) / np.maximum(count, 1), np.inf
    )
    used_g, used_p, dists = set(), set(), []
    for flat in np.argsort(mean, axis=None, kind="stable"):
        g, p = np.unravel_index(flat, mean.shape)
        if not np.isfinite(mean[g, p]) or g in used_g or p in used_p:
            continue
        used_g.add(g)
        used_p.add(p)
        dists.append(d[g, p][np.isfinite(d[g, p])])
    if not dists:
        return float("nan")
    return float(np.median(np.concatenate(dists)))

    def select_panel(
    groups: np.ndarray,
    names: Sequence[str],
    size: int = 64,
    hardness: Optional[np.ndarray] = None,
) -> List[Tuple[int, str]]:
    """A fixed, stratified panel of sample indices.

    Each group (validation file) gets an equal share of ``size`` (a group with fewer
    samples shows all of them and leaves its slots to the others). Within a group, a
    quarter of its share (at least one, when it has two or more slots and
    ``hardness`` is given) are its hardest samples (finite hardness only), the rest a
    seeded draw. The
    draw's seed is :func:`name_seed` of the group's name, so a group's picks do not
    depend on the other groups, the training seed or the process. With more groups
    than ``size``, ``size`` groups spread evenly over the list get one sample each.

    Args:
        groups: ``(n,)`` group (file) index of every sample.
        names: Group names, indexed by group.
        size: Maximum panel size.
        hardness: Optional ``(n,)`` score; higher is harder (ties: lower index).

    Returns:
        ``[(sample_index, role), ...]`` in display order: by group, then typical
        (by index) before hard (hardest first). ``role`` is ``"typical"`` or
        ``"hard"``.
    """
    groups = np.asarray(groups, dtype=int)
    if len(groups) == 0 or size <= 0:
        return []
    present = [g for g in range(len(names)) if np.any(groups == g)]
    if len(present) > size:
        keep = np.unique(np.linspace(0, len(present) - 1, size).round().astype(int))
        present = [present[i] for i in keep]
    members = [np.flatnonzero(groups == g) for g in present]
    quota = _quotas(np.array([len(m) for m in members]), size)

    panel = []
    for g, idx, q in zip(present, members, quota):
        if q <= 0:
            continue
        n_hard = 0
        if hardness is not None and q >= 2:
            n_hard = max(1, q // 4)
        hard = []
        if n_hard:
            h = np.asarray(hardness, dtype=np.float64)[idx]
            # Hardest first; ties keep the lower index (stable sort on -h). Samples
            # with no eligible box (hardness -inf) are never "hard".
            order = [i for i in np.argsort(-h, kind="stable") if np.isfinite(h[i])]
            hard = [int(idx[i]) for i in order[:n_hard]]
        rest = np.setdiff1d(idx, hard)
        rng = np.random.default_rng(name_seed(names[g]))
        n_typ = min(int(q) - len(hard), len(rest))
        typical = sorted(int(i) for i in rng.choice(rest, n_typ, replace=False))
        panel += [(i, "typical") for i in typical] + [(i, "hard") for i in hard]
    return panel

    def to_rgb_uint8(image) -> np.ndarray:
    """``(C, H, W)`` / ``(1, C, H, W)`` tensor or array -> ``(H, W, 3)`` uint8 RGB."""
    a = image.detach().cpu().numpy() if hasattr(image, "detach") else np.asarray(image)
    while a.ndim > 3:
        a = a[0]
    a = np.transpose(a, (1, 2, 0))
    if a.dtype != np.uint8:

        a = np.clip(a * scale, 0, 255).astype(np.uint8)

        a = np.repeat(a, 3, axis=-1)
    return np.ascontiguousarray(a[..., :3])
