    def export_model(
    model: torch.nn.Module,
    save_path: str | Path,
    fmt: str = "onnx",
    input_shape: Iterable[int] = (1, 1, 512, 512),
    opset_version: int = 17,
    output_names: Optional[list] = None,
    verify: bool = True,
    **kwargs,
) -> Path:
    """Export a model to the requested format."""
    fmt = fmt.lower()
    if fmt == "onnx":
        return export_to_onnx(
            model,
            save_path,
            input_shape=input_shape,
            opset_version=opset_version,
            output_names=output_names,
            verify=verify,
        )
    if fmt == "tensorrt":
        return export_to_tensorrt(model, save_path, input_shape=input_shape, **kwargs)
    if fmt == "both":
        export_to_onnx(
            model,
            save_path,
            input_shape=input_shape,
            opset_version=opset_version,
            output_names=output_names,
            verify=verify,
        )
        return export_to_tensorrt(model, save_path, input_shape=input_shape, **kwargs)

    raise ValueError(f"Unknown export format: {fmt}")

    def export_to_onnx(
    model: torch.nn.Module,
    save_path: str | Path,
    input_shape: Iterable[int] = (1, 1, 512, 512),
    input_dtype: torch.dtype = torch.uint8,
    opset_version: int = 17,
    dynamic_axes: Optional[Dict[str, Dict[int, str]]] = None,
    input_names: Optional[List[str]] = None,
    output_names: Optional[List[str]] = None,
    do_constant_folding: bool = True,
    verify: bool = True,
    numerical_check: bool = False,
    numerical_atol: float = 1e-3,
    numerical_rtol: float = 1e-3,
    static_hw: Optional[bool] = None,
) -> Path:
    """Export a PyTorch model to ONNX.

    Args:
        model: The PyTorch module to export.
        save_path: Destination path for the ``.onnx`` file.
        input_shape: Shape of the dummy input used to trace the graph.
        input_dtype: Dtype of the dummy input (``uint8`` matches the inference
            wrappers, which normalize in-graph).
        opset_version: ONNX opset for the (default) TorchScript exporter.
        dynamic_axes: Dynamic-axis spec; defaults to batch/height/width dynamic on
            the ``image`` input.
        input_names: ONNX input names; defaults to ``["image"]``.
        output_names: ONNX output names; inferred from a reference forward if
            ``None``.
        do_constant_folding: Whether to constant-fold during export.
        verify: If ``True``, run the structural ``onnx.checker`` on the result.
        numerical_check: If ``True``, additionally run the exported graph through
            onnxruntime on the export dummy input and assert output parity against
            PyTorch (atol/rtol below). ``_verify_onnx`` alone is structural only
            (``onnx.checker``); a numerical check catches graphs that are valid but
            wrong — a known failure mode for transformer backbones (e.g. Swin) whose
            ops trace but disagree numerically. Requires onnxruntime (the ``export``
            extra); degrades to a warning if it is unavailable.
        numerical_atol: Absolute tolerance for the numerical parity check.
        numerical_rtol: Relative tolerance for the numerical parity check.
        static_hw: If ``True``, export the input's height and width as static
            (the traced ``input_shape``); only the batch stays dynamic. ``None``
            (default) does so when the model contains a rotary position embedding
            (`has_rope`), whose grid tracing freezes. Pass ``False`` when the RoPE
            module only ever sees a fixed size anyway -- e.g. the instance model
            inside a top-down wrapper, which receives fixed-size crops.
    """
    save_path = Path(save_path)
    model.eval()

    if input_names is None:
        input_names = ["image"]
    if dynamic_axes is None:
        dynamic_axes = {"image": {0: "batch", 2: "height", 3: "width"}}
    rope = has_rope(model)
    if static_hw is None:
        static_hw = rope
    if static_hw:
        # A RoPE grid is only correct at the traced size. Export H/W static so
        # onnxruntime / TensorRT reject other sizes instead of returning silently
        # wrong outputs.
        dynamic_axes = _without_spatial_axes(dynamic_axes, input_names[0])
        reason = (
            "The backbone uses rotary position embeddings, which trace to a fixed "
            "token grid"
            if rope
            else "static_hw=True"
        )
        logger.warning(
            f"{reason}: exporting with static height/width "
            f"{tuple(input_shape)[2:]}; feed the ONNX model inputs of that size."
        )

    device = None
    try:
        device = next(model.parameters()).device
    except StopIteration:
        device = torch.device("cpu")

    if input_dtype.is_floating_point:
        dummy_input = torch.randn(*input_shape, device=device, dtype=input_dtype)
    else:
        dummy_input = torch.randint(
            0, 256, input_shape, device=device, dtype=input_dtype
        )

    # A reference forward is needed to infer output names and/or for parity.
    test_out = None
    if output_names is None or numerical_check:
        from sleap_nn.architectures.model import padded_like_export

        # The graph pads the network input to the backbone's max stride; so does
        # the reference.
        with torch.no_grad(), padded_like_export():
            test_out = model(dummy_input)
    if output_names is None:
        output_names = _infer_output_names(test_out)

    common = dict(
        input_names=input_names,
        output_names=output_names,
        dynamic_axes=dynamic_axes,
    )

    def _export_torchscript(opset: int) -> None:
        # Default to the legacy TorchScript exporter: it is fast and exports every
        # current wrapper (including the multi-stage top-down / multi-class ones that
        # the torch.export-based exporter cannot trace).
        torch.onnx.export(
            model,
            dummy_input,
            save_path.as_posix(),
            opset_version=opset,
            do_constant_folding=do_constant_folding,
            dynamo=False,
            **common,
        )

    torch.onnx.register_custom_op_symbolic(
        "aten::_upsample_bicubic2d_aa",
        _upsample_bicubic2d_aa_symbolic,
        _ANTIALIAS_OPSET,
    )
    try:
        _export_torchscript(opset_version)
    except torch.onnx.errors.UnsupportedOperatorError as err:
        try:
            if (
                opset_version >= _ANTIALIAS_OPSET
                or "_upsample_bicubic2d_aa" not in str(err)
            ):
                raise
            # A plain-ViT backbone's positional-embedding resample (antialiased
            # bicubic) lowers at opset 18 via the symbolic registered above. Stay on
            # the TorchScript exporter for it: it traces HF's shape arithmetic
            # dynamically, whereas torch.export pins the token grid (and, through
            # the [1, 3, 1, 1] mean/std broadcast, the batch) to the dummy input.
            _export_torchscript(_ANTIALIAS_OPSET)
        except torch.onnx.errors.UnsupportedOperatorError:
            # A few ops have no symbolic in the legacy exporter -- notably the
            # antialiased bilinear resize (``aten::_upsample_bilinear2d_aa``; the
            # single-instance wrapper builds its own from gathers instead, since
            # TensorRT parses no antialiased Resize). The torch.export-based
            # exporter supports them, so fall back to it for those models (it needs
            # opset >= 18 for the antialias Resize attribute).
            torch.onnx.export(
                model,
                dummy_input,
                save_path.as_posix(),
                opset_version=max(opset_version, _ANTIALIAS_OPSET),
                dynamo=True,
                **common,
            )

    if verify:
        _verify_onnx(save_path)

    if numerical_check:
        _verify_onnx_numerical(
            save_path,
            dummy_input,
            test_out,
            input_names[0],
            output_names,
            atol=numerical_atol,
            rtol=numerical_rtol,
        )

    return save_path

    Export a PyTorch model to TensorRT format.

    This function supports multiple compilation methods: - "onnx": Exports to ONNX first, then compiles with TensorRT (most reliable) - "jit": Uses torch.jit.trace + torch_tensorrt.compile (alternative)

    Parameters:

    Name Type Description Default
    model Module

    The PyTorch model to export (typically an ONNX wrapper).

    		 required
    save_path str | Path

    Path to save the TensorRT engine (.trt file).

    		 required
    input_shape Tuple[int, int, int, int]

    (B, C, H, W) optimal input tensor shape.

    		 (1, 1, 512, 512)
    input_dtype dtype

    Input tensor dtype (torch.uint8 or torch.float32).

    		 uint8
    precision str

    Model precision - "fp32", "tf32", "fp16" or "bf16" (ONNX method; the JIT method takes fp32 / tf32 / fp16).

    		 'fp16'
    min_shape Optional[Tuple[int, int, int, int]]

    Minimum input shape for dynamic shapes (default: batch=1, H/W halved).

    		 None
    opt_shape Optional[Tuple[int, int, int, int]]

    Optimal input shape (default: same as input_shape).

    		 None
    max_shape Optional[Tuple[int, int, int, int]]

    Maximum input shape (default: batch=16, H/W doubled).

    		 None
    workspace_size int

    TensorRT workspace size in bytes (default 2GB).

    		 2 << 30
    method str

    Compilation method - "onnx" or "jit".

    		 'onnx'
    verbose bool

    Print export info.

    		 True
    static_hw Optional[bool]

    Whether the engine takes only the traced height/width (a RoPE backbone's grid is fixed at trace time). None detects RoPE in model (has_rope); pass False for a wrapper whose RoPE model only sees fixed-size crops (a top-down instance model). For the "onnx" method an existing ONNX file's own static dims win.

    		 None
    _bf16_fp32_outside Optional[str]

    With precision="bf16": keep every float layer whose name does not contain this string (e.g. :data:BACKBONE_SCOPE) in fp32, so bf16 runs only inside that scope. None keeps only the transposed convolutions in fp32 (graphs whose outputs are feature maps, e.g. --network-only confidence maps).

    		 None
    _build_info Optional[dict]

    Optional dict filled with what was actually built: precision_built ("fp32" when fp16 fell back on a GPU without fast fp16), tensorrt_version, typing ("weak": builder flags; "strong": the precision carried by Cast nodes in the graph) and, for bf16, bf16_scope and bf16_fp32_layers.

    		 None
    force_precision bool

    Build fp16 / bf16 even for a model family whose reduced-precision engines do not reproduce its fp32 graph (multi-class bottom-up). Without it such a family is built in fp32 with a warning, on the strongly typed path (TensorRT 11, strongly_typed) and on TensorRT 10's builder flags alike.

    		 False
    strongly_typed bool

    TensorRT 10: build a strongly typed engine whose fp16 / bf16 is carried by Cast nodes in the graph (:func:~sleap_nn.export.exporters.onnx_precision.cast_onnx_precision), the way TensorRT 11 builds always. Default off: TensorRT 10 builds use the builder flags as before. Ignored by the "jit" method.

    		 False
    fp32_islands Optional[str]

    Recipe for precision="fp16" on the strongly typed path (TensorRT 11, or strongly_typed on 10): "attention_logits" keeps every attention block's logits product (Q.K^T, found by structure: :func:~sleap_nn.export.exporters.onnx_precision._find_attention_logits) in fp32, so TensorRT accumulates it in fp32; everything else is the usual fp16 rewrite. With _bf16_fp32_outside (graphs with coordinates) the fp16 runs only inside that scope, as bf16 does. Recorded as fp32_islands in the build info. "auto" (default): that recipe for an fp16 ONNX-method build of a model with a plain-ViT backbone (:func:has_plain_vit), whose plain fp16 engines are wrong (TensorRT's fp16 kernels accumulate Q.K^T in fp16); on TensorRT 10 such a build is then strongly typed. Any other build is unchanged. None: plain fp16 (as before for ViTs).

    		 'auto'

    Returns:

    Type Description
    Path

    Path to the exported TensorRT engine.
    Note

    TensorRT models are NOT cross-platform. The exported model will only work on the same GPU architecture and TensorRT version used for export.

    save_path: str | Path,
    input_shape: Tuple[int, int, int, int] = (1, 1, 512, 512),
    input_dtype: torch.dtype = torch.uint8,
    precision: str = "fp16",
    min_shape: Optional[Tuple[int, int, int, int]] = None,
    opt_shape: Optional[Tuple[int, int, int, int]] = None,
    max_shape: Optional[Tuple[int, int, int, int]] = None,
    workspace_size: int = 2 << 30,  # 2GB default
    method: str = "onnx",
    verbose: bool = True,
    static_hw: Optional[bool] = None,
    _bf16_fp32_outside: Optional[str] = None,
    _build_info: Optional[dict] = None,
    strongly_typed: bool = False,
    force_precision: bool = False,
    fp32_islands: Optional[str] = "auto",
) -> Path:
    """Export a PyTorch model to TensorRT format.

    This function supports multiple compilation methods:
    - "onnx": Exports to ONNX first, then compiles with TensorRT (most reliable)
    - "jit": Uses torch.jit.trace + torch_tensorrt.compile (alternative)

    Args:
        model: The PyTorch model to export (typically an ONNX wrapper).
        save_path: Path to save the TensorRT engine (.trt file).
        input_shape: (B, C, H, W) optimal input tensor shape.
        input_dtype: Input tensor dtype (torch.uint8 or torch.float32).
        precision: Model precision - "fp32", "tf32", "fp16" or "bf16" (ONNX
            method; the JIT method takes fp32 / tf32 / fp16).
        min_shape: Minimum input shape for dynamic shapes (default: batch=1, H/W halved).
        opt_shape: Optimal input shape (default: same as input_shape).
        max_shape: Maximum input shape (default: batch=16, H/W doubled).
        workspace_size: TensorRT workspace size in bytes (default 2GB).
        method: Compilation method - "onnx" or "jit".
        verbose: Print export info.
        static_hw: Whether the engine takes only the traced height/width (a RoPE
            backbone's grid is fixed at trace time). ``None`` detects RoPE in
            ``model`` (`has_rope`); pass ``False`` for a wrapper whose RoPE model
            only sees fixed-size crops (a top-down instance model). For the
            ``"onnx"`` method an existing ONNX file's own static dims win.
        _bf16_fp32_outside: With ``precision="bf16"``: keep every float layer whose
            name does not contain this string (e.g. :data:`BACKBONE_SCOPE`) in fp32,
            so bf16 runs only inside that scope. ``None`` keeps only the transposed
            convolutions in fp32 (graphs whose outputs are feature maps, e.g.
            ``--network-only`` confidence maps).
        _build_info: Optional dict filled with what was actually built:
            ``precision_built`` (``"fp32"`` when fp16 fell back on a GPU without
            fast fp16), ``tensorrt_version``, ``typing`` (``"weak"``: builder
            flags; ``"strong"``: the precision carried by Cast nodes in the graph)
            and, for bf16, ``bf16_scope`` and ``bf16_fp32_layers``.
        force_precision: Build fp16 / bf16 even for a model family whose
            reduced-precision engines do not reproduce its fp32 graph (multi-class
            bottom-up). Without it such a family is built in fp32 with a warning, on
            the strongly typed path (TensorRT 11, ``strongly_typed``) and on TensorRT
            10's builder flags alike.
        strongly_typed: TensorRT 10: build a strongly typed engine whose fp16 /
            bf16 is carried by Cast nodes in the graph
            (:func:`~sleap_nn.export.exporters.onnx_precision.cast_onnx_precision`),
            the way TensorRT 11 builds always. Default off: TensorRT 10 builds
            use the builder flags as before. Ignored by the ``"jit"`` method.
        fp32_islands: Recipe for ``precision="fp16"`` on the strongly typed path
            (TensorRT 11, or ``strongly_typed`` on 10): ``"attention_logits"`` keeps
            every attention block's logits product (Q.K^T, found by structure:
            :func:`~sleap_nn.export.exporters.onnx_precision._find_attention_logits`)
            in fp32, so TensorRT accumulates it in fp32; everything else is the
            usual fp16 rewrite. With ``_bf16_fp32_outside`` (graphs with coordinates)
            the fp16 runs only inside that scope, as bf16 does. Recorded as
            ``fp32_islands`` in the build info. ``"auto"`` (default): that recipe for
            an fp16 ONNX-method build of a model with a plain-ViT backbone
            (:func:`has_plain_vit`), whose plain fp16 engines are wrong (TensorRT's
            fp16 kernels accumulate Q.K^T in fp16); on TensorRT 10 such a build is
            then strongly typed. Any other build is unchanged. ``None``: plain fp16
            (as before for ViTs).

    Returns:
        Path to the exported TensorRT engine.

    Note:
        TensorRT models are NOT cross-platform. The exported model will only
        work on the same GPU architecture and TensorRT version used for export.
    """
    import tensorrt as trt

    model.eval()
    device = next(model.parameters()).device
    if str(precision).lower() not in ("fp32", "tf32", "fp16", "bf16"):
        raise ValueError(
            f"Unknown TensorRT precision {precision!r}: use fp32, tf32, fp16 or bf16."
        )
    precision = str(precision).lower()
    info = _build_info if _build_info is not None else {}
    auto_islands = fp32_islands == "auto"
    if auto_islands:
        fp32_islands = None  # decided below, once the precision is final
    if fp32_islands is not None:
        _check_fp32_islands(fp32_islands, precision, method, strongly_typed, trt)
    if precision in ("fp16", "bf16") and isinstance(model, _reduced_precision_fails()):
        # Built fp32 on every path (strongly typed or TensorRT 10's builder flags):
        # the reduced-precision engine of such a family is wrong, not just less exact.
        what = (
            f"a {type(model).__name__} engine does not reproduce its fp32 graph in "
            f"{precision} (multi-class bottom-up: 40-77% of peaks within 0.5 px)"
        )
        if not force_precision:
            msg = (
                f"Warning: {what}; building fp32 instead (force_precision to override)."
            )
            _warn_once(msg)
            info["precision_requested"] = precision
            info["precision_fallback"] = what + "; built fp32"
            precision = "fp32"
            if fp32_islands is not None:  # nothing left to keep in fp32
                info["fp32_islands"] = _islands_record(
                    fp32_islands, applied=False, reason="family built fp32"
                )
                fp32_islands = None
        else:
            msg = f"Warning: {what}; check the engine against fp32 on your data."
            _warn_once(msg)
    strong = bool(strongly_typed) or _uses_strong_typing(trt)
    if (
        auto_islands
        and precision == "fp16"
        and method == "onnx"
        and has_plain_vit(model)
    ):
        version = str(getattr(trt, "__version__", "0"))
        if not strong and _tensorrt_version(version) < _MIN_TENSORRT:
            # Strongly typed builds are supported from 10.13 only: keep the builder
            # flags, as promised for older versions, and say what the engine is.
            info["fp32_islands_mode"] = (
                f"not applied: TensorRT {version} is older than "
                f"{_MIN_TENSORRT[0]}.{_MIN_TENSORRT[1]}"
            )
            _warn_once(
                f"Warning: plain fp16 engine of a plain-ViT backbone (TensorRT {version} "
                "builds with its builder flags): its attention logits accumulate in "
                "fp16, so the engine may be inaccurate. Export with --precision bf16 or "
                f"fp32, or use TensorRT >= {_MIN_TENSORRT[0]}.{_MIN_TENSORRT[1]}."
            )
        elif not strong and not _platform_has_fast_fp16(trt, device):
            # The builder-flag build is fp32 on such a GPU (accurate): keep it.
            info["fp32_islands_mode"] = "not applied: no fast fp16 (built fp32)"
        else:
            fp32_islands = "attention_logits"
            info["fp32_islands_mode"] = "auto (plain-ViT backbone)"
            typed = ""
            if not strong:
                strongly_typed = True  # TensorRT 10's flags cannot keep a product fp32
                typed = ", strongly typed"
            _warn_once(
                "Note: fp16 engine of a plain-ViT backbone: each attention logits "
                f"product (Q.K^T) is kept in fp32{typed} (plain fp16 accumulates it in "
                "fp16, which moves keypoints and loses boxes). "
                "--no-fp32-attention-logits (fp32_islands=None) builds plain fp16."
            )
    elif fp32_islands is not None:
        info["fp32_islands_mode"] = "requested"
        if not strong:
            strongly_typed = True  # the recipe needs the graph's types
            _warn_once(
                "Note: fp32 attention logits: building strongly typed (TensorRT 10's "
                "builder flags cannot keep single products in fp32)."
            )

    save_path = Path(save_path)
    if not save_path.suffix:
        save_path = save_path.with_suffix(".trt")

    B, C, H, W = input_shape

    if min_shape is None:
        min_shape = (1, C, H // 2, W // 2)
    if opt_shape is None:
        opt_shape = input_shape
    if max_shape is None:
        max_shape = (min(16, B * 4), C, H * 2, W * 2)
    if static_hw is None:
        static_hw = has_rope(model)
    if method == "jit" and static_hw:
        # The traced RoPE grid is only correct at the traced size (the ONNX route
        # gets the same restriction from the static ONNX input instead).
        pinned = tuple((s[0], s[1], H, W) for s in (min_shape, opt_shape, max_shape))
        if pinned != (tuple(min_shape), tuple(opt_shape), tuple(max_shape)):
            logger.warning(
                "The backbone uses rotary position embeddings, which trace to a "
                f"fixed token grid: TensorRT profile height/width pinned to {H}x{W}."
            )
        min_shape, opt_shape, max_shape = pinned

    if verbose:
        print(f"Exporting model to TensorRT...")
        print(f"  Input shape: {input_shape}")
        print(f"  Min/Opt/Max: {min_shape} / {opt_shape} / {max_shape}")
        print(f"  Precision: {precision}")
        print(f"  Workspace: {workspace_size / 1e9:.1f} GB")
        print(f"  Method: {method}")

    if method == "onnx":
        # Build on the model's GPU (--device cuda:N), and judge its precision support
        # there, not on whichever device is current.
        cuda_ctx = (
            torch.cuda.device(device)
            if device.type == "cuda" and torch.cuda.is_available()
            else nullcontext()
        )
        with cuda_ctx:
            return _export_tensorrt_onnx(
                model,
                save_path,
                input_shape,
                input_dtype,
                min_shape,
                opt_shape,
                max_shape,
                precision,
                workspace_size,
                verbose,
                static_hw,
                _bf16_fp32_outside=_bf16_fp32_outside,
                _build_info=info,
                strongly_typed=strongly_typed,
                fp32_islands=fp32_islands,
            )
    elif method == "jit":
        return _export_tensorrt_jit(
            model,
            save_path,
            input_shape,
            input_dtype,
            min_shape,
            opt_shape,
            max_shape,
            precision,
            workspace_size,
            verbose,
        )
    else:
        raise ValueError(f"Unknown method: {method}. Use 'onnx' or 'jit'.")
